function setupAuth(){
  onAuthStateChanged(auth, async (user) => {
    if(user){
      currentUser = user;
      document.getElementById('loginBtnWrap').style.display = 'none';
      document.getElementById('logoutBtn').style.display = 'block';
      document.getElementById('userName').innerText = user.displayName || 'ログインユーザー';
      if(user.photoURL){
        document.getElementById('userAvatar').innerHTML = `<img src="${user.photoURL}" alt="avatar">`;
      }
      
      const uDoc = await getDoc(doc(db, 'users', user.uid));
      if(uDoc.exists()){
        const ud = uDoc.data();
        if(ud.themeColor) applyThemeColor(ud.themeColor);
        if(ud.splashImg) applySplashImage(ud.splashImg, false);
        if(ud.homeIcon) applyHomeIcon(ud.homeIcon, false);
      }
      subscribeEvents(user.uid);
    }else{
      currentUser = null;
      document.getElementById('loginBtnWrap').style.display = 'block';
      document.getElementById('logoutBtn').style.display = 'none';
      document.getElementById('userName').innerText = 'ゲストユーザー';
      document.getElementById('userAvatar').innerHTML = '<i class="ti ti-user"></i>';
      if(unsubEvents) { unsubEvents(); unsubEvents = null; }
      loadLocalEvents();
    }
  });
}

}

  const mobInp = document.getElementById('mobSearchInput');

function filterEvents(list){
  return list.filter(e => {
    if(selectedTagFilter!=='all' && e.tag!==selectedTagFilter) return false;
    if(currentSearchQuery){
      const t = (e.title||'').toLowerCase();
      const l = (e.location||'').toLowerCase();
      const m = (e.memo||'').toLowerCase();
      if(!t.includes(currentSearchQuery) && !l.includes(currentSearchQuery) && !m.includes(currentSearchQuery)) return false;
    }
    return true;
  });
}

// カレンダー描画（PC・スマホ対応）
function buildCalendar(){
  const grid = document.getElementById('calGrid');
  if(!grid) return;
  
  const firstDay = new Date(currentYear, currentMonth, 1).getDay();
  const lastDate = new Date(currentYear, currentMonth + 1, 0).getDate();
  const prevLastDate = new Date(currentYear, currentMonth, 0).getDate();
  
  let cells = [];
  // 前月
  for(let i=firstDay-1; i>=0; i--){
    const y = currentMonth===0?currentYear-1:currentYear;
    const m = currentMonth===0?11:currentMonth-1;
    cells.push({dateStr: dStr(y,m,prevLastDate-i), num: prevLastDate-i, isOther: true});
  }
  // 当月
  for(let i=1; i<=lastDate; i++){
    cells.push({dateStr: dStr(currentYear, currentMonth, i), num: i, isOther: false});
  }
  // 翌月（42マス一律にする）
  const nextLen = 42 - cells.length;
  for(let i=1; i<=nextLen; i++){
    const y = currentMonth===11?currentYear+1:currentYear;
    const m = currentMonth===11?0:currentMonth+1;
    cells.push({dateStr: dStr(y,m,i), num: i, isOther: true});
  }
  
  const fList = filterEvents(events);
  const tStr = todayStr();
  
  grid.innerHTML = cells.map((c, idx) => {
    const dayOfWeek = idx % 7;
    let classes = ['day-cell'];
    if(c.isOther) classes.push('other');
    if(c.dateStr === tStr) classes.push('today');
    if(c.dateStr === selectedDate) classes.push('selected');
    if(dayOfWeek === 0) classes.push('sun');
    if(dayOfWeek === 6) classes.push('sat');
    
    // 祝日簡易判定（振替対応等なし、固定のみお掃除用最小限）
    let holLabel = '';
    const md = c.dateStr.substring(5);
    if(md==='01-01'){holLabel='元日'; classes.push('holiday');}
    else if(md==='05-03'){holLabel='憲法記念日'; classes.push('holiday');}
    else if(md==='05-04'){holLabel='みどりの日'; classes.push('holiday');}
    else if(md==='05-05'){holLabel='こどもの日'; classes.push('holiday');}
    else if(md==='11-03'){holLabel='文化の日'; classes.push('holiday');}
    else if(md==='11-23'){holLabel='勤労感謝の日'; classes.push('holiday');}
    
    const dayEvs = fList.filter(e => e.date === c.dateStr).sort((a,b)=>(a.time||'00:00').localeCompare(b.time||'00:00'));
    
    const evsHtml = dayEvs.map(e => {
      const tObj = TAGS.find(x => x.id === e.tag) || TAGS[0];
      return `<div class="mini-event" style="background:${tObj.color}" onclick="openDetailModal('${e.id}', event)">${e.time?e.time+' ':''}${e.title}</div>`;
    }).join('');
    
    return `
      <div class="${classes.join(' ')}" data-date="${c.dateStr}" onclick="handleDayCellClick('${c.dateStr}')">
        <div>
          <span class="dnum">${c.num}</span>
          ${holLabel?`<span class="h-label">${holLabel}</span>`:''}
        </div>
        <div class="cell-events">${evsHtml}</div>
      </div>
    `;
  }).join('');
  
  renderMobCalEventList();
}

          <span style="color:${tObj.color};font-weight:600">${tObj.name}</span>

// 一覧ビュー描画
function renderList(){
  const container = document.getElementById('listContainer');
  if(!container) return;
  
  const fList = filterEvents(events).sort((a,b) => a.date.localeCompare(b.date) || (a.time||'00:00').localeCompare(b.time||'00:00'));
  
  if(fList.length === 0){
    container.innerHTML = '<div style="text-align:center;color:var(--text2);padding:40px 0"><i class="ti ti-calendar-off" style="font-size:32px;margin-bottom:8px;display:block"></i>該当するイベントが見つかりません</div>';
    return;
  }
  
  // 月ごとにグループ化
  let groups = {};
  fList.forEach(e => {
    const key = e.date.substring(0,7); // "2026-06"
    if(!groups[key]) groups[key] = [];
    groups[key].push(e);
  });
  
  let html = '';
  Object.keys(groups).sort().forEach(k => {
    const parts = k.split('-');
    html += `
      <div class="list-section">
        <div class="list-section-title">${parts[0]}年${parts[1]}月</div>
        <div class="list-events-grid">
          ${groups[k].map(e => buildEventCardHtml(e)).join('')}
        </div>
      </div>
    `;
  });
  container.innerHTML = html;
}

window.addChekiRow = function(){
  const nameInp = document.getElementById('chekiNameInp');
  const priceInp = document.getElementById('chekiPriceInp');
  const countInp = document.getElementById('chekiCountInp');
  
  const name = nameInp.value.trim();
  const price = parseInt(priceInp.value) || 0;
  const count = parseInt(countInp.value) || 0;
  
  if(!name){ alert('メンバー名を入力してください'); return; }
  if(count <= 0){ alert('枚数は1箇所以上を指定してください'); return; }
  
  const existing = temporaryChekis.find(x => x.name === name && x.price === price);
  if(existing){
    existing.count += count;
  }else{
    temporaryChekis.push({id: Date.now().toString(), name, price, count});
  }
  
  nameInp.value = '';
  countInp.value = '1';
  renderFormChekiList();
};

window.deleteChekiRow = function(id){
  temporaryChekis = temporaryChekis.filter(x => x.id !== id);
  renderFormChekiList();
};

function renderFormChekiList(){
  const list = document.getElementById('formChekiList');
  const total = document.getElementById('chekiTotalSum');
  if(!list || !total) return;
  
  let sum = 0;
  list.innerHTML = temporaryChekis.map(c => {
    const rowSub = c.price * c.count;
    sum += rowSub;
    return `
      <div class="cheki-item">
        <div class="cheki-item-name" title="${c.name}">${c.name}</div>
        <div class="cheki-item-num">¥${c.price}</div>
        <div class="cheki-item-num">${c.count}枚</div>
        <button class="cheki-del-btn" onclick="deleteChekiRow('${c.id}')"><i class="ti ti-trash"></i></button>
      </div>
    `;
  }).join('');
  total.innerText = sum.toLocaleString();
}

window.handleSaveEvent = async function(){
  const id = document.getElementById('formEventId').value;
  const title = document.getElementById('formTitle').value.trim();
  const date = document.getElementById('formDate').value;
  const time = document.getElementById('formTime').value;
  const location = document.getElementById('formLocation').value.trim();
  const price = document.getElementById('formPrice').value;
  const memo = document.getElementById('formMemo').value;
  const tag = document.getElementById('tagSelectContainer').dataset.selected || 'live';
  
  if(!title || !date){ alert('イベント名と日付は必須です'); return; }
  
  const payload = { title, date, time, location, price, memo, tag, chekis: temporaryChekis, updatedAt: Date.now() };
  
  setSyncStatus('syncing');
  if(currentUser){
    try{
      if(id){ await updateDoc(doc(db, 'users', currentUser.uid, 'events', id), payload); }
      else { await addDoc(collection(db, 'users', currentUser.uid, 'events'), payload); }
      showToast('クラウドに保存しました');
    }catch(e){ console.error(e); setSyncStatus('err'); alert('通信エラーが発生しました'); }
  }else{
    if(id){
      const idx = events.findIndex(x => x.id === id);
      if(idx!==-1) events[idx] = {id, ...payload};
    }else{
      events.push({id: Date.now().toString(), ...payload});
    }
    saveLocalEvents();
    showToast('ローカルに保存しました');
    refreshUI();
  }
  closeEventModal();
  closeDetailModal();
};

window.handleDeleteEvent = async function(){
  const id = document.getElementById('formEventId').value;
  if(!id) return;
  if(!confirm('このイベントを削除しますか？')) return;
  
  setSyncStatus('syncing');
  if(currentUser){
    try{
      await deleteDoc(doc(db, 'users', currentUser.uid, 'events', id));
      showToast('イベントを削除しました');
    }catch(e){ console.error(e); setSyncStatus('err'); }
  }else{
    events = events.filter(x => x.id !== id);
    saveLocalEvents();
    showToast('イベントを削除しました');
    refreshUI();
  }
  closeEventModal();
  closeDetailModal();
};
